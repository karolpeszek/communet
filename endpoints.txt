GET /api/v1/vehicle/<vehicle uuid>/getinfo
//return serialized json data containing these fields:
- uuid
- license plate number
- type (string with possible options: TRAM, BUS, TRAIN)
- owner (operator, eg. "MPK Krakow")


GET /api/v1/stop/<stop uuid>/getinfo
//return serialized json data containing these fields:
- uuid
- longitude and latitude
- short_name (like "Czerwone Maki")
- long_name (like "Czerone Maki P+R 3t")
- zone_id (like "KRAKOW_STREFA_1")
- type (corresponds to the vehicle type, possible values are TRAM, BUS, TRAIN)


GET /api/v1/route/<route uuid>/getinfo
//return serialized json data containing these fields:
-uuid
-line number (as string) (eg. "192" or "A8" or "SKA1")
-destination (eg "Wieliczka Miasto")
-vehicles (list of uuids of the vehicles operating on a particular route)
-stops (list of uuids of the stops in order on current route)

GET /api/v1/trip/<trip uuid>/getinfo
//return serialized json data containng these felds
-uuid
-route_uuid - the route that trip operates on
-timestamps (list of timestamps matching in length with the list of stops in current route)
//timestamps should be unix milisecond timestamps

POST /api/v1/planning/getroutes
data input:
{
    "start": {
        "latitude": 1234,
        "longitude": 5678
    },
    "destination": {
        "latitude": 2137,
        "longitude": 69420
    },
    "journey_start_timestamp": (unix timestamp ms)
}

[
    //list of available journeys:
    {
        //data for one jurney
        "start_stop_uuid": "", //uuid of the starting stop
        "destination_stop_uuid": "", //uuid of the destination stop
        "journey_start_timestamp": "", //timestamp of the beginning of the journey
        "journey_duration": 2137, // journey duration in minutes
        connections: [
            //following connections
            {
                "trip_uuid": "" //uuid of the particular trip
                "start_stop_uuid": //uuid of the beginning stop
                "stop_stop_uuid": //uuid of the destionation stop for that connection
                "vehicle_uuid": //uuid of the handling vehicle

            }
        ]



    }
